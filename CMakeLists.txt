cmake_minimum_required(VERSION 3.21)
project(ViscousMotion_Lab_5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SFML 3 CONFIG REQUIRED COMPONENTS Graphics Window System)
add_executable(ViscousMotion_Lab_5 ViscousMotion_Lab_5/ViscousMotion_Lab_5.cpp)
target_link_libraries(ViscousMotion_Lab_5 PRIVATE SFML::Graphics SFML::Window SFML::System)
if(WIN32)
 add_custom_command(TARGET ViscousMotion_Lab_5 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:ViscousMotion_Lab_5> $<TARGET_FILE_DIR:ViscousMotion_Lab_5> COMMAND_EXPAND_LISTS)
endif()
